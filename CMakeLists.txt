cmake_minimum_required(VERSION 3.12)

project(Remmy)

# communication layer
if (NOT COMM_LAYER)
    message(WARNING "Communication layer is not specified, using ASIO as default. Currently you can use ASIO or ZMQ.")
    set(COMM_LAYER "ASIO" CACHE STRING "Choose the type of communication layer." FORCE)
endif(NOT COMM_LAYER)

if(COMM_LAYER STREQUAL "ASIO")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -DUSE_ASIO=1 -DASIO_STANDALONE=1")
elseif(COMM_LAYER STREQUAL "ZMQ")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -DUSE_ZMQ=1")
    link_directories(deps/zmq)
else()
    message(FATAL_ERROR "Currently only ASIO and ZMQ are supported as communication layer.")
endif(COMM_LAYER STREQUAL "ASIO")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Threads REQUIRED)

function(configure_remmy_target target)
    target_include_directories(${target} PRIVATE ${CMAKE_SOURCE_DIR})
    target_link_libraries(${target} PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(${target} PRIVATE /Zc:preprocessor)
    endif()
    if(COMM_LAYER STREQUAL "ASIO")
        target_include_directories(${target} SYSTEM PRIVATE deps/asio/asio/include)
    elseif(COMM_LAYER STREQUAL "ZMQ")
        target_include_directories(${target} SYSTEM PRIVATE deps/zmq)
        target_link_libraries(${target} PRIVATE zmq)
    endif()
endfunction()

add_executable(remmy_test test/test.cpp)
configure_remmy_target(remmy_test)

include(CTest)
if(BUILD_TESTING)
    foreach(suite serialize sleeplist)
        add_executable(${suite}_test test/${suite}_test.cpp)
        configure_remmy_target(${suite}_test)
        add_test(NAME ${suite} COMMAND ${suite}_test)
        set_tests_properties(${suite} PROPERTIES TIMEOUT 15)
    endforeach()
    add_test(NAME rpc_example COMMAND remmy_test)
    set_tests_properties(rpc_example PROPERTIES TIMEOUT 20)
    if(COMM_LAYER STREQUAL "ASIO")
        add_executable(asio_test test/asio_test.cpp)
        configure_remmy_target(asio_test)
        add_test(NAME asio_bad_first_header COMMAND asio_test first)
        add_test(NAME asio_bad_subsequent_header COMMAND asio_test subsequent)
        add_executable(rpc_test test/rpc_test.cpp)
        configure_remmy_target(rpc_test)
        add_test(NAME rpc COMMAND rpc_test)
        set_tests_properties(asio_bad_first_header asio_bad_subsequent_header rpc
            PROPERTIES TIMEOUT 20)
    endif()
endif()
